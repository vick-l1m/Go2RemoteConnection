# Lives in /etc/systemd/system/go2-remote-connection.service

[Unit]
Description=Go2 Remote Connection startup service
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=unitree
Group=unitree

# Folder that contains your startup script
WorkingDirectory=/home/unitree/go2_ws/Go2RemoteConnection

# Run your startup script
# Change "joystick" to "terminal" or "movement" if you want a different mode
ExecStart=/bin/bash /home/unitree/go2_ws/Go2RemoteConnection/RL_start_remote_connection.sh joystick

# Restart if it crashes
Restart=always
RestartSec=5

# Useful environment
Environment=HOME=/home/unitree
Environment=PYTHONUNBUFFERED=1

# Send logs to journalctl
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
